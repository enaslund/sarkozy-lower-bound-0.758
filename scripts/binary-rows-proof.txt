/-- All fractional-power endpoints are proved by finite integer square tests. -/
private theorem row_power_lower (i : Fin @COUNT@) :
    (rowLower i : ℝ) ≤ (rowScale i : ℝ)^(componentPowers 8) := by
  have h := rpow_lower_of_integer_ladders @DEN@ 154942497274 1000000000000
    (rowLowerLadder i) (rowUpperLadder i) (by decide) (by decide)
    (row_lower_positive i) (row_upper_positive i) (row_lower_steps i)
    (row_upper_steps i) (row_endpoints i)
  rw [row_first_lower, row_first_upper, Nat.cast_mul, Nat.cast_mul] at h
  have hl : ((rowScaleNumerator i:ℝ)*@LFACTOR@)/@DEN@ =
      (rowScaleNumerator i:ℝ)/@GRID@ := by ring
  have hu : ((rowLowerNumerator i:ℝ)*@UFACTOR@)/@DEN@ =
      (rowLowerNumerator i:ℝ)/@POWERS@ := by ring
  norm_num only [Nat.cast_ofNat] at h
  rw [hl,hu] at h
  have hf : componentPowers 8 = (154942497274/1000000000000:ℝ) := rfl
  rw [hf]
  norm_num only [rowScale, rowLower, Rat.cast_div, Rat.cast_natCast]
  exact h

private theorem row_scale_below (s : Fin 25) (r : ℤ) (hr : r ∈ policy.branches s) :
    (rowScale (rowIndex s r):ℝ) ≤ policy.scale s r := by
  have hb := policy_valid.branch s r hr
  let j : Fin 4 := ⟨r.toNat, by omega⟩
  have hj : (j:ℤ) = r := by dsimp [j]; omega
  have h := row_scale_rational s j
    (by simpa only [hj, policy, RationalBinaryPolicy.toReal] using hr)
  rw [hj] at h
  exact Rat.cast_le.mpr h

      ((∑ r ∈ rationalPolicy.branches s,
        rowLower (rowIndex s r)*rationalVector (rationalPolicy.child s r):ℚ):ℝ) :=
    Rat.cast_le.mpr (row_sum_rational s)
  push_cast at hrat
  change growth*vector s ≤ ∑ r ∈ policy.branches s,
